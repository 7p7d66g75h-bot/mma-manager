 function setLive(x){let el=document.getElementById("live");if(el)el.textContent=x}
 function hit(att,def,attF){
   let zone=Math.random(),dmg=0,label="";
   let hand=(attF.handPower||attF.power||70),leg=(attF.legPower||attF.power||70);
   let accuracy=clamp(((attF.handSpeed||attF.speed||70)+(attF.movement||attF.speed||70))/220,.55,.88);
   if(Math.random()>accuracy){return "не попадает"}
   if(zone<.45){dmg=rand(0,2)+Math.max(0,Math.round((hand-88)/55));def.head-=dmg;label="в голову"}
   else if(zone<.75){dmg=rand(0,2)+Math.max(0,Math.round((hand-90)/60));def.body-=dmg;label="в корпус"}
   else{dmg=rand(0,2)+Math.max(0,Math.round((leg-90)/60));def.legs-=dmg;label="по ногам"}
   dmg=Math.max(1,dmg);att.score+=dmg;att.roundScore+=dmg;att.sig++;
   att.hits=(att.hits||0)+1;
   return label;
 }
 function event(){
   let edge=(ovr(f)-ovr(o))/120+(f.form-o.form)/320+(a.stamina-b.stamina)/500;
   let t=Math.random();
   if(t<.62){
     if(Math.random()+edge*.35>.5){a.strikes=(a.strikes||0)+1;let label=hit(a,b,f);setLive(`${f.name} точно попадает ${label}`)}
     else{b.strikes=(b.strikes||0)+1;let label=hit(b,a,o);setLive(`${o.name} отвечает ${label}`)}
   }else if(t<.76){
     let tdEdge=((f.takedown||f.wrest)- (o.takedown||o.wrest))/180;
     if(a.td<3&&Math.random()+tdEdge>.62){a.td++;a.control+=rand(2,5);a.roundScore+=rand(2,5);b.stamina=Math.max(0,b.stamina-rand(4,8));setLive(`${f.name} забирает тейкдаун и контроль`)}
     else if(b.td<3&&Math.random()-tdEdge>.58){b.td++;b.control+=rand(2,5);b.roundScore+=rand(2,5);a.stamina=Math.max(0,a.stamina-rand(4,8));setLive(`${o.name} проводит тейкдаун`)}
     else setLive("Клинч у сетки — борьба за позицию")
   }else{

 }

   win=true;method="Submission";finishLabel="SUBMISSION!";
 }else if(zoneKO||zoneKO2){
   win=zoneKO&&!zoneKO2;method="KO/TKO";finishLabel="KO!";
 }else{
   // Only a completed scheduled fight can reach the decision branch.
   const validRounds=scorecards.length;
   const totals=scorecards.slice(0,validRounds).reduce((t,r)=>({a:t.a+r.a,b:t.b+r.b}),{a:0,b:0});
   const aTotal=totals.a,bTotal=totals.b;
   const aWins=scorecards.filter(r=>r.a>r.b).length;
   const bWins=scorecards.filter(r=>r.b>r.a).length;

   if(aWins>bWins){win=true;method="Decision";finishLabel="WINNER — UNANIMOUS DECISION";}
   else if(bWins>aWins){win=false;method="Decision";finishLabel="WINNER — UNANIMOUS DECISION";}
   else{
     // Genuine split decision: two judges give it to one side, one to the other.
     method="Split Decision";
     const winnerA=aTotal>=bTotal;
     win=winnerA;
     finishLabel="WINNER — SPLIT DECISION";
   }
   scoreLines=scorecards.map(r=>`${r.a}-${r.b}`);
